run() { # id file sed-expr tests...
  id=$1; f=$2; expr=$3; shift 3
  cp "$f" "$S/dm-$id.bak"
  sed -i "$expr" "$f"
  if cmp -s "$f" "$S/dm-$id.bak"; then echo "$id: SED DID NOT CHANGE THE FILE"; return; fi

  printf "%s -> " "$id"; grep -E '^# (pass|fail) ' "$S/dm-$id.txt" | tr '\n' ' '; echo
  grep -E '^    not ok' "$S/dm-$id.txt" | head -1 | cut -c1-170
  cp "$S/dm-$id.bak" "$f"
  cmp "$f" "$S/dm-$id.bak" && echo "$id: restored byte for byte"
}
